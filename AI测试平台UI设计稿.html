# tests/test_auth_service.py
import pytest
from unittest.mock import patch, MagicMock
from service.auth_service import AuthService

class TestAuthServiceLogin:
  def setup_method(self):
    self.auth_service = AuthService()

  @patch('dao.user_dao.UserDAO.get_user_by_email')
  @patch('utils.password_utils.PasswordUtils.verify_password')
  def test_login_success(self, mock_verify, mock_get_user):
    # Given
    mock_get_user.return_value = {'id': 1, 'email': 'test@example.com', 'password_hash': 'hashed'}
    mock_verify.return_value = True

    # When
    result = self.auth_service.login('test@example.com', 'correct_password')

    # Then
    assert result['user_id'] == 1
    assert 'token' in result

  @patch('dao.user_dao.UserDAO.get_user_by_email')
  def test_login_user_not_found(self, mock_get_user):
    mock_get_user.return_value = None
    with pytest.raises(UserNotFoundError):
      self.auth_service.login('notfound@example.com', 'any_password')
